let inputConfig = input.config();
let recordId = inputConfig.rec;

let mixerTable = base.getTable("Mixer Cycle");
let dailyTankTable = base.getTable("Daily Resin Tank");

// Fetch the specific Mixer Cycle record
let targetMixerRecord = await mixerTable.selectRecordAsync(recordId, {
    fields: [
        "M1_R_DTN", "M2_R_DTN", "M3_R_DTN", "M4_R_DTN",
        "M1_R_W", "M2_R_W", "M3_R_W", "M4_R_W"
    ]
});

if (!targetMixerRecord) {
    output.set("error", "Record not found in Mixer Cycle");
    return;
}

// Helper functions
let extractName = (obj) => obj?.name || null;
let extractWeight = (val) => (typeof val === "number" ? val : 0);

// Fields mapping
let mixerFields = ["M1_R_DTN", "M2_R_DTN", "M3_R_DTN", "M4_R_DTN"];
let resinIdFields = ["M1_R_ID", "M2_R_ID", "M3_R_ID", "M4_R_ID"];
let weightFields = ["M1_R_W", "M2_R_W", "M3_R_W", "M4_R_W"];

// Aggregate weights by Tank Name
let weightSums = {};
for (let i = 0; i < mixerFields.length; i++) {
    let name = extractName(targetMixerRecord.getCellValue(mixerFields[i]));
    let weight = extractWeight(targetMixerRecord.getCellValue(weightFields[i]));
    if (name) weightSums[name] = (weightSums[name] || 0) + weight;
}

// Fetch all Daily Resin Tank records
let dailyTankRecords = await dailyTankTable.selectRecordsAsync({
    fields: ["Daily Tank No.", "Remaining Weight"]
});

// Organize tank data
let tankData = {};
dailyTankRecords.records.forEach(record => {
    let name = extractName(record.getCellValue("Daily Tank No."));
    if (!name) return;

    let remainingWeight = extractWeight(record.getCellValue("Remaining Weight"));

    if (!tankData[name]) tankData[name] = [];
    tankData[name].push({ id: record.id, remainingWeight });
});

// Match and consume from tanks (progressively)
let matchedTanks = {}; // { "DTN-1": [ { id, remainingWeight, usedWeight }, ... ] }
for (let [name, totalWeight] of Object.entries(weightSums)) {
    let sortedTanks = (tankData[name] || []).sort((a, b) => a.remainingWeight - b.remainingWeight);
    let remainingToDeduct = totalWeight;
    let usedTanks = [];

    for (let tank of sortedTanks) {
        if (remainingToDeduct <= 0) break;

        let usedWeight = Math.min(remainingToDeduct, tank.remainingWeight);
        usedTanks.push({
            id: tank.id,
            usedWeight,
            remainingWeight: tank.remainingWeight
        });

        remainingToDeduct -= usedWeight;
    }

    if (usedTanks.length > 0 && remainingToDeduct <= 0) {
        matchedTanks[name] = usedTanks;
    }
}

// Prepare Mixer Cycle update — link first matching tank only
let mixerUpdates = {};
for (let i = 0; i < mixerFields.length; i++) {
    let name = extractName(targetMixerRecord.getCellValue(mixerFields[i]));
    if (name && matchedTanks[name]?.length > 0) {
        mixerUpdates[resinIdFields[i]] = [{ id: matchedTanks[name][0].id }];
    }
}

// Update Mixer Cycle
if (Object.keys(mixerUpdates).length > 0) {
    await mixerTable.updateRecordAsync(recordId, mixerUpdates);
} else {
    output.set("error", "No tanks had enough combined weight to fulfill mixer needs.");
    return;
}

// Prepare Daily Resin Tank updates
let tankUpdates = [];
for (let usedTanks of Object.values(matchedTanks)) {
    for (let { id, remainingWeight, usedWeight } of usedTanks) {
        tankUpdates.push({
            id,
            fields: {
                "Remaining Weight": remainingWeight - usedWeight
            }
        });
    }
}

// Function to update records in batches of 50
async function updateInBatches(table, records) {
    const batchSize = 50;
    for (let i = 0; i < records.length; i += batchSize) {
        const batch = records.slice(i, i + batchSize);
        await table.updateRecordsAsync(batch);
    }
}

// Update Daily Resin Tank Remaining Weights in batches
if (tankUpdates.length > 0) {
    await updateInBatches(dailyTankTable, tankUpdates);
}

// Output matched tank info for reference
output.set("matchedTanks", matchedTanks);
