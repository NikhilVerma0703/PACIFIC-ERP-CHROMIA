// ===============================
// Step 1: Inputs & table handles
// ===============================

let inputConfig = input.config();
let inputSlabNumber = inputConfig.inputSlabNumber;   // May be undefined on hourly trigger
let createdRecordId = inputConfig.recordId;          // May be undefined on hourly trigger

console.log("Step 1: inputSlabNumber:", inputSlabNumber, "recordId:", createdRecordId);

let changeTable = base.getTable("Change Parameters Oven");
let ovenTable = base.getTable("Oven");
console.log("Step 2: Tables loaded: 'Change Parameters Oven' and 'Oven'");

// Fetch all parameter records
let changeRecords = await changeTable.selectRecordsAsync();
console.log("Step 3: Fetched records count from 'Change Parameters Oven':", changeRecords.records.length);

// Fetch all Oven records
let ovenRecords = await ovenTable.selectRecordsAsync();
let ovenFieldsSet = new Set(ovenTable.fields.map(f => f.name));
console.log("Step 4: Fetched records from 'Oven':", ovenRecords.records.length);

// ===============================
// Helper functions
// ===============================

// Helper A: Find best matching parameter record
function findBestMatch(targetSlab, changeRecords) {
    if (typeof targetSlab !== "number" || isNaN(targetSlab)) return null;

    let bestMatch = null;
    let closestSlab = -Infinity;

    for (let record of changeRecords.records) {
        let slab = record.getCellValue("Effective from slab number");
        if (typeof slab === "number" && slab <= targetSlab) {
            if (slab > closestSlab) {
                closestSlab = slab;
                bestMatch = record;
            }
        }
    }
    return bestMatch;
}

// Helper B: Calculate Cooking Time (in minutes)
function calculateCookingTime(record) {
    let inTime = record.getCellValue("In Time");
    let outTime = record.getCellValue("Out Time");

    if (inTime !== null && outTime !== null) {
        let inSeconds = inTime;
        let outSeconds = outTime;
        let durationMinutes;

        if (outSeconds >= inSeconds) {
            durationMinutes = Math.round((outSeconds - inSeconds) / 60);
        } else {
            // Out Time is on the next day
            durationMinutes = Math.round((24 * 3600 - inSeconds + outSeconds) / 60);
        }
        return durationMinutes;
    }
    return null;
}

// Helper C: Prepare updates for a single Oven record
function prepareUpdatesForOven(ovenRecord, parameterRecord) {
    let updates = {};

    // 1. Copy parameters from Change Parameters table if empty
    if (parameterRecord) {
        for (let field of changeTable.fields) {
            // Skip control fields
            if (
                field.name === "Effective from slab number" ||
                field.name === "Effective from slab count"
            ) {
                continue;
            }

            if (ovenFieldsSet.has(field.name)) {
                let currentVal = ovenRecord.getCellValue(field.name);
                let newVal = parameterRecord.getCellValue(field.name);

                if (newVal !== null && currentVal === null) {
                    updates[field.name] = newVal;
                }
            }
        }
    }

    // 2. Calculate Cooking Time if empty
    let currentCooking = ovenRecord.getCellValue("Cooking Time");
    if (currentCooking === null) {
        let calculatedTime = calculateCookingTime(ovenRecord);
        if (calculatedTime !== null) {
            updates["Cooking Time"] = calculatedTime;
        }
    }

    return updates;
}

// ===============================
// PART 1: Process current record
// (only if recordId + slab are provided)
// ===============================

if (createdRecordId && typeof createdRecordId === "string") {
    console.log("Part 1: Processing triggered record:", createdRecordId);

    // Parse slab from input; if missing/NaN, skip Part 1 gracefully
    let currentSlabNum =
        typeof inputSlabNumber === "number"
            ? inputSlabNumber
            : parseFloat(inputSlabNumber);

    if (!isNaN(currentSlabNum)) {
        let currentMatchedParam = findBestMatch(currentSlabNum, changeRecords);

        if (currentMatchedParam) {
            console.log(
                "Step 5: Found match for current record. Slab:",
                currentMatchedParam.getCellValue("Effective from slab number")
            );
        } else {
            console.log("Step 5: No matching parameter record found for current slab.");
        }

        // Get the actual oven record object safely
        let matchingOvenRecord = null;
        try {
            matchingOvenRecord = ovenRecords.getRecord(createdRecordId);
        } catch (e) {
            console.log(
                "Step 6: Created Oven record not found in current query. ID:",
                createdRecordId
            );
        }

        if (matchingOvenRecord) {
            let updates = prepareUpdatesForOven(matchingOvenRecord, currentMatchedParam);

            if (Object.keys(updates).length > 0) {
                await ovenTable.updateRecordAsync(createdRecordId, updates);
                console.log("Step 8: Updated current Oven record.");
            } else {
                console.log("Step 8: No updates needed for current record.");
            }
        } else {
            console.log("Step 6: Skipping Part 1 because record was not found in ovenRecords.");
        }
    } else {
        console.log(
            "Step 5: inputSlabNumber is missing or invalid; skipping Part 1 for this run."
        );
    }
} else {
    console.log(
        "Part 1: No createdRecordId provided (likely scheduled trigger); skipping current-record processing."
    );
}

// ===============================
// PART 2: Global backfill
// ===============================

console.log("🔄 Starting Global Backfill for unfilled Oven records...");

let updatesToProcess = [];

// Optional safety cap to avoid timeouts in very large bases
const MAX_RECORDS_PER_RUN = 700;

for (let ovenRec of ovenRecords.records) {
    // Skip the record we may have just processed in Part 1
    if (createdRecordId && ovenRec.id === createdRecordId) continue;

    let slabNumRaw = ovenRec.getCellValue("Slab Number");
    let slabNum = typeof slabNumRaw === "number" ? slabNumRaw : parseFloat(slabNumRaw);

    // Skip if no slab number (can't match parameters without it)
    if (isNaN(slabNum)) continue;

    // Check if this record is "incomplete" (key fields empty)
    let isMissingCooking = ovenRec.getCellValue("Cooking Time") === null;

    // If fully complete (by your key metric), skip
    if (!isMissingCooking) continue;

    // Find parameters for this historical record
    let historicalParam = findBestMatch(slabNum, changeRecords);

    // Calculate potential updates
    let backfillUpdates = prepareUpdatesForOven(ovenRec, historicalParam);

    if (Object.keys(backfillUpdates).length > 0) {
        updatesToProcess.push({
            id: ovenRec.id,
            fields: backfillUpdates,
        });

        // Respect per-run safety cap
        if (updatesToProcess.length >= MAX_RECORDS_PER_RUN) {
            console.log(
                `Reached MAX_RECORDS_PER_RUN (${MAX_RECORDS_PER_RUN}); remaining records will be handled in future runs.`
            );
            break;
        }
    }
}

// Execute batch updates (max 50 at a time, per Airtable limits)
if (updatesToProcess.length > 0) {
    console.log(`⚡ Found ${updatesToProcess.length} historical Oven records to backfill.`);

    while (updatesToProcess.length > 0) {
        let batch = updatesToProcess.splice(0, 50);
        await ovenTable.updateRecordsAsync(batch);
        console.log(`...processed batch of ${batch.length}`);
    }
    console.log("✅ All backfill updates completed for this run.");
} else {
    console.log(
        "ℹ️ No historical backfill needed (no incomplete records matched criteria this run)."
    );
}
