/***********************
 * STEP 1: Get input
 ***********************/
let inputVars = input.config();
let siloInputRaw = inputVars["Silo No"];
console.log("Raw Silo No input:", siloInputRaw);

/***********************
 * STEP 2: Resolve Silo No safely
 ***********************/
let siloNoInput = null;

if (typeof siloInputRaw === "string") {
    siloNoInput = siloInputRaw;
} else if (Array.isArray(siloInputRaw)) {
    if (siloInputRaw.length > 0 && siloInputRaw[0]?.name) {
        siloNoInput = siloInputRaw[0].name;
    } else {
        throw new Error("'Silo No' input array is empty or invalid.");
    }
} else if (siloInputRaw && typeof siloInputRaw === "object" && siloInputRaw.name) {
    siloNoInput = siloInputRaw.name;
} else {
    throw new Error("'Silo No' is missing or not usable.");
}

console.log("Resolved Silo No:", siloNoInput);

/***********************
 * STEP 3: Query SILO table
 ***********************/
let siloTable = base.getTable("SILO");
let siloQuery = await siloTable.selectRecordsAsync();

let matchingRecords = siloQuery.records.filter(
    r => r.getCellValueAsString("Silo No") === siloNoInput
);

if (matchingRecords.length === 0) {
    throw new Error(`No matching Silo No found in SILO table for "${siloNoInput}"`);
}

/***********************
 * STEP 4: Query Silo Emptying Log
 ***********************/
let siloEmptyingLogTable = base.getTable("Silo Emptying Log");
let emptyingLogQuery = await siloEmptyingLogTable.selectRecordsAsync();

let matchingLogRecord = emptyingLogQuery.records.find(
    r => r.getCellValueAsString("Silo No") === siloNoInput
);

if (!matchingLogRecord) {
    throw new Error(`No matching Silo Emptying Log found for "${siloNoInput}"`);
}

/***********************
 * STEP 5: Bag Weight (default = 600 if empty)
 ***********************/
let bagWeightRaw = matchingLogRecord.getCellValue("Bag Weight");

let DEFAULT_BAG_WEIGHT = 600;
let bagWeight = null;

if (bagWeightRaw === null || bagWeightRaw === undefined) {
    bagWeight = DEFAULT_BAG_WEIGHT;
    console.warn(`Bag Weight missing → defaulting to ${DEFAULT_BAG_WEIGHT} kg`);
} else if (typeof bagWeightRaw === "number") {
    bagWeight = bagWeightRaw;
} else if (typeof bagWeightRaw === "string") {
    bagWeight = Number(bagWeightRaw);
} else if (bagWeightRaw?.value !== undefined) {
    bagWeight = Number(bagWeightRaw.value);
}

if (!Number.isFinite(bagWeight) || bagWeight <= 0) {
    throw new Error(`Invalid Bag Weight: ${JSON.stringify(bagWeightRaw)}`);
}

/***********************
 * STEP 6: Empty ID
 ***********************/
let emptyIDFromLog = matchingLogRecord.getCellValue("Empty ID");

if (!emptyIDFromLog) {
    throw new Error("Empty ID missing in Silo Emptying Log.");
}

console.log("Bag Weight used:", bagWeight);
console.log("Empty ID:", emptyIDFromLog);

/***********************
 * STEP 7: Extract Grade from SILO
 ***********************/
let gradeValue = null;

for (let r of matchingRecords) {
    let g = r.getCellValue("Grade (from Used Bag)");
    if (!gradeValue && Array.isArray(g) && g.length > 0) {
        gradeValue = g[0];
    }
}

console.log("Grade transferred:", gradeValue);

/***********************
 * STEP 8: Supplier = Unloaded
 ***********************/
let supplierTable = base.getTable("Supplier Master");
let supplierQuery = await supplierTable.selectRecordsAsync();

let supplierRecord = supplierQuery.records.find(
    r => r.getCellValueAsString("Name") === "Unloaded"
);

if (!supplierRecord) {
    throw new Error("Supplier 'Unloaded' not found.");
}

let unloadedSupplierRef = [{ id: supplierRecord.id }];

/***********************
 * STEP 9: RM table setup
 ***********************/
let rmTable = base.getTable("RM");

let typeRef = { name: "Filler" };
let sizeRef = { name: "400 #" };

let gradeField = rmTable.getField("Grade");
let gradeValueToUse = null;

if (gradeValue && gradeValue.id) {
    // RM → Grade is a linked record field
    gradeValueToUse = [{ id: gradeValue.id }];
}


/***********************
 * STEP 10: Duplicate protection
 ***********************/
let existingRMQuery = await rmTable.selectRecordsAsync();

let duplicate = existingRMQuery.records.find(
    r => r.getCellValueAsString("Empty ID") === emptyIDFromLog.toString()
);

if (duplicate) {
    throw new Error(`RM already exists for Empty ID ${emptyIDFromLog}`);
}

/***********************
 * STEP 11: Total silo weight
 ***********************/
let totalSiloWeight = 0;

for (let r of matchingRecords) {
    let w = r.getCellValue("Remaining Weight");
    if (typeof w === "number" && w > 0) {
        totalSiloWeight += w;
    }
}

console.log("Total remaining silo weight:", totalSiloWeight);

/***********************
 * STEP 12: Bag calculation
 ***********************/
let totalBags = Math.floor(totalSiloWeight / bagWeight);
let totalWeightUsed = totalBags * bagWeight;

if (totalBags < 1) {
    throw new Error("Not enough material for even one bag.");
}

console.log(`Creating ${totalBags} bags, total ${totalWeightUsed} kg`);

/***********************
 * STEP 13: Create RM records
 ***********************/
let recordsToCreate = [];

for (let i = 1; i <= totalBags; i++) {
    let fields = {
        "Bag Weight": bagWeight,
        "Supplier Master": unloadedSupplierRef,
        "Type": typeRef,
        "Size": sizeRef,
        "INV No": "Unloaded",
        "Bag No": i,
        "Empty ID": emptyIDFromLog.toString()
    };

    if (gradeValueToUse) {
        fields["Grade"] = gradeValueToUse;
    }

    recordsToCreate.push({ fields });
}

while (recordsToCreate.length > 0) {
    await rmTable.createRecordsAsync(recordsToCreate.splice(0, 50));
}

/***********************
 * STEP 14: Deduct silo weight (distributed)
 ***********************/
let weightLeft = totalWeightUsed;
let siloUpdates = [];

for (let r of matchingRecords) {
    if (weightLeft <= 0) break;

    let current = r.getCellValue("Remaining Weight") || 0;
    if (current <= 0) continue;

    let newWeight = Math.max(0, current - weightLeft);
    weightLeft -= current;

    siloUpdates.push({
        id: r.id,
        fields: { "Remaining Weight": newWeight }
    });
}

while (siloUpdates.length > 0) {
    await siloTable.updateRecordsAsync(siloUpdates.splice(0, 50));
}

console.log("✅ SCRIPT COMPLETED SUCCESSFULLY");
