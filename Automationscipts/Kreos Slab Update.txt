// Step 1: Get tables
let changeTable = base.getTable("Change Parameters Kreos");
let targetTable = base.getTable("Kreos");
console.log("Step 1: Tables loaded");

// Step 2: Get all records from both tables
let [changeRecords, targetRecords] = await Promise.all([
    changeTable.selectRecordsAsync(),
    targetTable.selectRecordsAsync()
]);
console.log("Step 2: Records fetched - Params:", changeRecords.records.length, "Kreos:", targetRecords.records.length);

// Step 3: Build sorted list of parameter records
let paramRecordsSorted = changeRecords.records
    .filter(r => typeof r.getCellValue("Effective from slab number") === "number")
    .sort((a, b) => b.getCellValue("Effective from slab number") - a.getCellValue("Effective from slab number"));

// Step 4: Cache fields in Kreos for comparison
let targetFieldNames = new Set(targetTable.fields.map(f => f.name));

// Step 5: Update only incomplete Kreos records
for (let kreosRecord of targetRecords.records) {
    let slabNumber = kreosRecord.getCellValue("Slab Number");
    if (typeof slabNumber !== "number") continue;

    // Find best matching parameter record
    let matchedParam = paramRecordsSorted.find(param => {
        let effectiveFrom = param.getCellValue("Effective from slab number");
        return slabNumber >= effectiveFrom;
    });

    if (!matchedParam) {
        console.log(`No parameter found for slab ${slabNumber}`);
        continue;
    }

    let fieldsToUpdate = {};

    for (let field of changeTable.fields) {
        let fieldName = field.name;
        let fieldType = field.type;

        // Skip control/incompatible fields
        if (
            ["Effective from slab number", "Effective from slab count", "Operator"].includes(fieldName) ||
            ["formula", "createdTime", "lastModifiedTime", "multipleAttachments", "multipleLookupValues"].includes(fieldType)
        ) {
            continue;
        }

        // Proceed only if field exists in Kreos
        if (targetFieldNames.has(fieldName)) {
            let existing = kreosRecord.getCellValue(fieldName);
            let paramVal = matchedParam.getCellValue(fieldName);

            if (paramVal !== null && (existing === null || existing === "")) {
                fieldsToUpdate[fieldName] = paramVal;
            }
        }
    }

    // Apply updates if needed
    if (Object.keys(fieldsToUpdate).length > 0) {
        console.log(`Updating slab ${slabNumber} (${kreosRecord.id})`);
        await targetTable.updateRecordAsync(kreosRecord.id, fieldsToUpdate);
    } else {
        console.log(`No updates needed for slab ${slabNumber}`);
    }
}

console.log("All applicable Kreos records updated.");
