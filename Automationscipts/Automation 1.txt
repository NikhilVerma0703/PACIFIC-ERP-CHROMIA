// ---------- CONFIG ----------
const tableName = "Slab Segregation"; // Replace with your actual table name
const linkedFieldName = "Slab Numbers QC"; // Field used for splitting linked records
const textFieldName = "Slab Number";       // Field used for splitting space-separated slab numbers
const autoNumberFieldName = "SNo";         // AutoNumber field

// Lookup fields to skip
const lookupFields = [
    "Shade Photo (from batch no)",
    "Shade Description (from batch no)",
    "Shade type (from batch no)"
];

// ---------- SCRIPT ----------

// Get triggering record
let inputConfig = input.config();
let recordId = inputConfig.recordId;

let table = base.getTable(tableName);
let record = await table.selectRecordAsync(recordId);

if (!record) {
    output.text("No record found for ID: " + recordId);
    return;
}

// Decide which field to split
let slabNumbersQC = record.getCellValue(linkedFieldName); // Linked records array
let slabNumberText = record.getCellValue(textFieldName);  // Space-separated text

let slabList = [];

// If linked field is filled
if (slabNumbersQC && slabNumbersQC.length > 0) {
    slabList = slabNumbersQC.map(linked => ({ type: "linked", value: linked.id }));
}
// Else if text field is filled
else if (slabNumberText && slabNumberText.trim() !== "") {
    slabList = slabNumberText
        .split(" ")
        .map(num => num.trim())
        .filter(num => num !== "")
        .map(num => ({ type: "text", value: num }));
} else {
    output.text("Neither 'Slab Numbers QC' nor 'Slab Number' is filled.");
    return;
}

// Build a list of editable fields to copy
let fieldsToCopy = {};
for (let field of table.fields) {
    let fieldName = field.name;

    // Skip fields that cannot be copied
    if (
        fieldName === linkedFieldName ||      // Splitting linked field
        fieldName === textFieldName ||        // Splitting text field
        fieldName === autoNumberFieldName ||  // AutoNumber field
        lookupFields.includes(fieldName)      // Lookup fields
    ) {
        continue; // Skip these
    }

    // Copy value
    fieldsToCopy[fieldName] = record.getCellValue(fieldName);
}

// Create new records
let newRecords = slabList.map(item => {
    let newFields = { ...fieldsToCopy };

    if (item.type === "linked") {
        newFields[linkedFieldName] = [{ id: item.value }];
    } else if (item.type === "text") {
        newFields[textFieldName] = item.value;
    }

    return { fields: newFields };
});

// Batch create records (max 50 per batch)
while (newRecords.length > 0) {
    await table.createRecordsAsync(newRecords.slice(0, 50));
    newRecords = newRecords.slice(50);
}

console.log(`Created ${slabList.length} new records successfully.`);

// Delete the original record
await table.deleteRecordAsync(recordId);
console.log("Parent record deleted successfully.");
