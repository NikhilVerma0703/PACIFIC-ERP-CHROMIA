/**
 * PRODUCTION-READY: Grit Silo Allocation System
 * FINAL CORRECTED FIELD NAMES - CONFIRMED
 * 
 * ⚠️ IMPORTANT: This code ONLY processes Grit fields (M1-M4)
 * It does NOT touch Filler fields and won't interfere with allocateFillerSilo()
 * 
 * FINAL CONFIRMED FIELD NAMING:
 * - M1_W1, M1_W2, ..., M4_W5 (WEIGHT fields)
 * - M1_G1_SN, M1_G2_SN, ..., M4_G5_SN (SILO NAME fields)
 * - M1_G1, M1_G2, ..., M4_G5 (OUTPUT LINK fields)
 * 
 * Smart Delink Logic:
 * - Only delinks if LATER entries are linked BUT EARLIER entries are not
 * - Prevents unnecessary refunds and reallocations
 * - FIFO priority is maintained
 * 
 * Fields IGNORED (FILLER - NOT TOUCHED):
 * - M{1-4}_F_W
 * - Filler_Silo/Buffer
 * - Filler Silo ID
 */

// ============================================================================
// UTILITY FUNCTIONS
// ============================================================================

function getMatchValue(cellValue) {
    if (!cellValue) return null;
    if (Array.isArray(cellValue)) return cellValue[0]?.name?.trim() || null;
    if (typeof cellValue === "object" && cellValue.name) return cellValue.name.trim();
    return String(cellValue).trim();
}

function parseBatchNumber(batchStr) {
    if (!batchStr) return Infinity;
    const numPart = String(batchStr).replace(/\D/g, "");
    return parseInt(numPart) || Infinity;
}

// ============================================================================
// MAIN ALLOCATION FUNCTION - GRIT ONLY (FINAL CORRECTED FIELD NAMES)
// ============================================================================

async function allocateGritSilos() {
    const mixerTable = base.getTable("Mixer Cycle");
    const siloTable = base.getTable("SILO");

    console.log("\n" + "=".repeat(70));
    console.log("🚀 STARTING: Grit Silo Allocation (M1-M4, G1-G5)");
    console.log("⚠️  NOTE: Filler fields will NOT be affected");
    console.log("=".repeat(70));

    try {
        console.log("📦 Fetching mixer records...");
        const mixerRecords = await mixerTable.selectRecordsAsync();
        console.log(`   → Found ${mixerRecords.records.length} mixer records`);

        console.log("📦 Fetching silo records...");
        const siloRecords = await siloTable.selectRecordsAsync();
        console.log(`   → Found ${siloRecords.records.length} silo records`);

        // ========================================================================
        // PHASE 0: CHECK IF DELINK/REFUND NEEDED (SMART LOGIC)
        // ========================================================================

        console.log("\n📋 Phase 0: Checking if delink/refund needed...");

        const needsDelink = new Map(); // Track which silos need delinking

        // First pass: collect all demands sorted by batch/cycle
        const allDemands = [];
        for (const record of mixerRecords.records) {
            const batch = record.getCellValue("Batch");
            const batchNum = parseBatchNumber(batch);
            const cycle = record.getCellValue("Cycle") || 0;

            for (let m = 1; m <= 4; m++) {
                for (let g = 1; g <= 5; g++) {
                    // FINAL CORRECTED FIELD NAMES
                    const weightFieldName = `M${m}_W${g}`;
                    const siloFieldName = `M${m}_G${g}_SN`;
                    const linkFieldName = `M${m}_G${g}`;

                    const weight = record.getCellValue(weightFieldName) || 0;
                    const siloNo = getMatchValue(record.getCellValue(siloFieldName));
                    const existingLink = record.getCellValue(linkFieldName);

                    if (!weight || !siloNo) continue;

                    allDemands.push({
                        recordId: record.id,
                        siloNo,
                        weight,
                        batchNum,
                        cycle,
                        linkFieldName,
                        weightFieldName,
                        siloFieldName,
                        materialGrade: `M${m}G${g}`,
                        hasLink: existingLink && existingLink.length > 0
                    });
                }
            }
        }

        // Sort by FIFO to find out correct order
        allDemands.sort((a, b) => {
            if (a.batchNum !== b.batchNum) return a.batchNum - b.batchNum;
            return a.cycle - b.cycle;
        });

        // Check if any later entry has link while earlier doesn't
        for (const demand of allDemands) {
            if (demand.hasLink) {
                // Found a linked entry, check if any earlier entry for same silo is unlinked
                const earlierDemands = allDemands.filter(d => 
                    d.siloNo === demand.siloNo &&
                    (d.batchNum < demand.batchNum || 
                     (d.batchNum === demand.batchNum && d.cycle < demand.cycle))
                );

                const hasUnlinkedEarlier = earlierDemands.some(d => !d.hasLink && d.weight > 0);

                if (hasUnlinkedEarlier) {
                    if (!needsDelink.has(demand.siloNo)) {
                        needsDelink.set(demand.siloNo, []);
                    }
                    needsDelink.get(demand.siloNo).push(demand);
                }
            }
        }

        if (needsDelink.size === 0) {
            console.log("   ✅ No delink needed (no FIFO conflicts detected)");
        } else {
            console.log(`   ⚠️  Found FIFO conflicts: ${needsDelink.size} silo(s) need reallocation`);
        }

        // ========================================================================
        // PHASE 1: BUILD SILO MAP
        // ========================================================================

        console.log("\n📋 Phase 1: Building silo map...");

        const siloMap = {};
        for (const silo of siloRecords.records) {
            const siloNo = getMatchValue(silo.getCellValue("Silo No"))?.toLowerCase();
            if (!siloNo) continue;
            if (!siloMap[siloNo]) siloMap[siloNo] = [];
            silo._remainingWeight = silo.getCellValue("Remaining Weight") || 0;
            siloMap[siloNo].push(silo);
        }

        // Sort by increment within each silo
        for (const key in siloMap) {
            siloMap[key].sort((a, b) => (a.getCellValue("SILO Increment") || 0) - (b.getCellValue("SILO Increment") || 0));
        }

        console.log(`   → Built map for ${Object.keys(siloMap).length} silo(s)`);

        // ========================================================================
        // PHASE 2: REFUND LINKED SILOS (ONLY IF NEEDED)
        // ========================================================================

        console.log("\n📋 Phase 2: Refunding linked silos (only if delink needed)...");

        const refundMap = new Map();
        const mixerDelinkedMap = new Map();
        let refundCount = 0;

        if (needsDelink.size > 0) {
            for (const record of mixerRecords.records) {
                for (let m = 1; m <= 4; m++) {
                    for (let g = 1; g <= 5; g++) {
                        const linkFieldName = `M${m}_G${g}`;
                        const weightFieldName = `M${m}_W${g}`;
                        const siloFieldName = `M${m}_G${g}_SN`;

                        const linkedSilos = record.getCellValue(linkFieldName);
                        const weight = record.getCellValue(weightFieldName) || 0;
                        const siloName = getMatchValue(record.getCellValue(siloFieldName));

                        // Only refund if this silo needs delinking
                        if (linkedSilos && linkedSilos.length > 0 && weight > 0 && needsDelink.has(siloName?.toLowerCase())) {
                            let refundAmount = weight;
                            for (const linkedSilo of linkedSilos) {
                                if (refundAmount <= 0) break;

                                const actualSilo = siloRecords.records.find(s => s.id === linkedSilo.id);
                                if (actualSilo) {
                                    const silo = actualSilo;
                                    const siloNo = getMatchValue(silo.getCellValue("Silo No"))?.toLowerCase();

                                    if (!refundMap.has(silo.id)) {
                                        refundMap.set(silo.id, {
                                            id: silo.id,
                                            siloNo: siloNo,
                                            amount: 0
                                        });
                                    }

                                    const refundThisSilo = Math.min(refundAmount, weight / linkedSilos.length);
                                    refundMap.get(silo.id).amount += refundThisSilo;
                                    silo._remainingWeight += refundThisSilo;
                                    refundAmount -= refundThisSilo;
                                }
                            }

                            // Mark for delinking
                            if (!mixerDelinkedMap.has(record.id)) {
                                mixerDelinkedMap.set(record.id, { id: record.id, fields: {} });
                            }

                            mixerDelinkedMap.get(record.id).fields[linkFieldName] = [];
                            console.log(`   ↩️  Refunded ${weight}kg from ${siloName} → M${m}G${g}`);
                            refundCount++;
                        }
                    }
                }
            }

            console.log(`   → Refunded ${refundCount} entries from ${refundMap.size} silos`);
        } else {
            console.log("   → No refunds needed (skipped delink)");
        }

        // ========================================================================
        // PHASE 3: DELINK ENTRIES (ONLY IF NEEDED)
        // ========================================================================

        console.log("\n📋 Phase 3: Delinking entries (only if delink needed)...");

        if (mixerDelinkedMap.size > 0) {
            const delinkedUpdates = Array.from(mixerDelinkedMap.values());
            let batchCount = 0;
            while (delinkedUpdates.length > 0) {
                const batch = delinkedUpdates.slice(0, 50);
                await mixerTable.updateRecordsAsync(batch);
                batchCount++;
                delinkedUpdates.splice(0, 50);
            }
            console.log(`   → Delinked ${mixerDelinkedMap.size} mixer records`);
        } else {
            console.log("   → No delinks needed (skipped)");
        }

        // ========================================================================
        // PHASE 4: COLLECT ALL UNLINKED DEMANDS
        // ========================================================================

        console.log("\n📋 Phase 4: Collecting unlinked grit demands...");

        const unlinkedDemands = [];
        for (const record of mixerRecords.records) {
            const batch = record.getCellValue("Batch");
            const batchNum = parseBatchNumber(batch);
            const cycle = record.getCellValue("Cycle") || 0;

            for (let m = 1; m <= 4; m++) {
                for (let g = 1; g <= 5; g++) {
                    const weightFieldName = `M${m}_W${g}`;
                    const siloFieldName = `M${m}_G${g}_SN`;
                    const linkFieldName = `M${m}_G${g}`;

                    const weight = record.getCellValue(weightFieldName) || 0;
                    const siloNo = getMatchValue(record.getCellValue(siloFieldName));
                    const existingLink = record.getCellValue(linkFieldName);

                    // Skip if: no weight | no silo | already linked
                    if (!weight || !siloNo || (existingLink && existingLink.length > 0)) {
                        continue;
                    }

                    unlinkedDemands.push({
                        recordId: record.id,
                        siloNo,
                        weight,
                        batchNum,
                        cycle,
                        linkFieldName,
                        materialGrade: `M${m}G${g}`
                    });
                }
            }
        }

        console.log(`   → Total unlinked demands: ${unlinkedDemands.length}`);

        if (unlinkedDemands.length === 0) {
            console.log("\n   → No unlinked demands to fulfill.");
            console.log("=".repeat(70));
            console.log("✨ Grit allocation complete! (No changes made)");
            console.log("=".repeat(70) + "\n");
            return;
        }

        // ========================================================================
        // PHASE 5: SORT BY FIFO
        // ========================================================================

        console.log("\n📋 Phase 5: Sorting by FIFO order...");

        unlinkedDemands.sort((a, b) => {
            if (a.batchNum !== b.batchNum) return a.batchNum - b.batchNum;
            return a.cycle - b.cycle;
        });

        console.log(`   Sample (first 3):`);
        unlinkedDemands.slice(0, 3).forEach(d => {
            console.log(`   B${d.batchNum}C${d.cycle} → ${d.materialGrade}: ${d.weight}kg from "${d.siloNo}"`);
        });

        // ========================================================================
        // PHASE 6: ALLOCATE (SINGLE PASS) - PRODUCTION-READY FIX
        // ========================================================================

        console.log("\n📋 Phase 6: Allocating grit in FIFO order...");

        const siloUpdateMap = new Map();
        const mixerUpdateMap = new Map();

        let fulfilledCount = 0;
        let unfulfilledCount = 0;

        for (const demand of unlinkedDemands) {
            const siloKey = demand.siloNo.toLowerCase();
            const silos = siloMap[siloKey];

            if (!silos || silos.length === 0) {
                console.log(`   ❌ ${demand.materialGrade} [B${demand.batchNum}C${demand.cycle}]: Silo "${demand.siloNo}" not found`);
                unfulfilledCount++;
                continue;
            }

            let weightLeft = demand.weight;
            const used = [];
            
            // 🔥 PRODUCTION FIX: Temporary staging prevents partial allocation bug
            const tempSiloUpdates = new Map();
            let tempWeightLeft = weightLeft;

            // Stage allocation without committing
            for (const silo of silos) {
                if (tempWeightLeft <= 0) break;

                const avail = silo._remainingWeight;
                if (avail <= 0) continue;

                const take = Math.min(avail, tempWeightLeft);
                const newRemaining = avail - take;
                
                tempWeightLeft -= take;
                used.push({ id: silo.id });

                // Stage update only - do NOT modify main maps or memory yet
                tempSiloUpdates.set(silo.id, {
                    id: silo.id,
                    fields: { "Remaining Weight": newRemaining },
                    _newVal: newRemaining
                });
            }

            // ✅ Only commit if FULL demand satisfied (PRODUCTION SAFETY)
            if (tempWeightLeft === 0 && used.length > 0) {
                
                // Queue mixer link update
                if (!mixerUpdateMap.has(demand.recordId)) {
                    mixerUpdateMap.set(demand.recordId, {
                        id: demand.recordId,
                        fields: {}
                    });
                }
                const mixerUpdate = mixerUpdateMap.get(demand.recordId);
                mixerUpdate.fields[demand.linkFieldName] = [...used];

                // Commit staged silo updates to main map
                for (const [siloId, updateData] of tempSiloUpdates) {
                    siloUpdateMap.set(siloId, {
                        id: updateData.id,
                        fields: updateData.fields
                    });
                    
                    // Update in-memory object for subsequent iterations
                    const siloObj = silos.find(s => s.id === siloId);
                    if (siloObj) {
                        siloObj._remainingWeight = updateData._newVal;
                    }
                }

                console.log(`   ✅ ${demand.materialGrade} [B${demand.batchNum}C${demand.cycle}]: FULFILLED (${demand.weight}kg)`);
                fulfilledCount++;
            } else {
                console.log(`   ⚠️  ${demand.materialGrade} [B${demand.batchNum}C${demand.cycle}]: UNFULFILLED (need ${tempWeightLeft.toFixed(1)}kg more)`);
                unfulfilledCount++;
            }
        }

        // ========================================================================
        // PHASE 7: DATABASE UPDATES
        // ========================================================================

        console.log("\n📋 Phase 7: Applying database updates (grit only)...");

        // Update silos first
        const siloUpdates = Array.from(siloUpdateMap.values());
        console.log(`   📦 Updating ${siloUpdates.length} silo records...`);
        while (siloUpdates.length > 0) {
            await siloTable.updateRecordsAsync(siloUpdates.slice(0, 50));
            siloUpdates.splice(0, 50);
        }

        // Update mixer records (GRIT only)
        const mixerUpdates = Array.from(mixerUpdateMap.values());
        console.log(`   📦 Updating ${mixerUpdates.length} mixer records (grit fields only)...`);
        while (mixerUpdates.length > 0) {
            await mixerTable.updateRecordsAsync(mixerUpdates.slice(0, 50));
            mixerUpdates.splice(0, 50);
        }

        // ========================================================================
        // PHASE 8: SUMMARY
        // ========================================================================

        console.log("\n" + "=".repeat(70));
        console.log("📊 GRIT ALLOCATION SUMMARY");
        console.log("=".repeat(70));
        console.log(`✅ Fulfilled: ${fulfilledCount}/${unlinkedDemands.length}`);
        console.log(`❌ Unfulfilled: ${unfulfilledCount}/${unlinkedDemands.length}`);
        console.log(`📝 Silos updated: ${siloUpdateMap.size}`);
        console.log(`📝 Mixer grit records updated: ${mixerUpdateMap.size}`);
        if (needsDelink.size > 0) {
            console.log(`♻️  Reallocation: ${refundCount} entries relinked`);
        }
        console.log(`⚠️  Filler fields were NOT modified`);
        console.log("=".repeat(70));
        console.log("✨ Grit allocation complete!");
        console.log("=".repeat(70) + "\n");

    } catch (error) {
        console.error("\n❌ FATAL ERROR in Grit Allocation");
        console.error(`Error: ${error.message}`);
        console.error(`Stack: ${error.stack}`);
        throw error;
    }
}

// ============================================================================
// EXECUTE
// ============================================================================

await allocateGritSilos();


// 🔄 Filler Silo FIFO Allocation with deduped batching

function getMatchValue(cellValue) {
    if (!cellValue) return null;
    if (Array.isArray(cellValue)) return cellValue[0]?.name?.trim() || null;
    if (typeof cellValue === "object" && cellValue.name) return cellValue.name.trim();
    return String(cellValue).trim();
}

function parseBatchNumber(batchStr) {
    if (!batchStr) return Infinity;
    const numPart = String(batchStr).replace(/\D/g, "");
    return parseInt(numPart) || Infinity;
}

async function allocateFillerSilo() {
    const mixerTable = base.getTable("Mixer Cycle");
    const siloTable = base.getTable("SILO");

    const mixerRecords = await mixerTable.selectRecordsAsync();
    const siloRecords = await siloTable.selectRecordsAsync();

    // 📦 Collect all unlinked demands
    const fillerDemands = [];
    for (const record of mixerRecords.records) {
        const fillerObj = record.getCellValue("Filler_Silo/Buffer");
        const siloNo = getMatchValue(fillerObj);
        const batch = record.getCellValue("Batch");
        const batchNum = parseBatchNumber(batch);
        const cycle = record.getCellValue("Cycle") || 0;
        const link = record.getCellValue("Filler Silo ID");

        const weights = [
            record.getCellValue("M1_F_W") || 0,
            record.getCellValue("M2_F_W") || 0,
            record.getCellValue("M3_F_W") || 0,
            record.getCellValue("M4_F_W") || 0
        ];
        const totalWeight = weights.reduce((a, b) => a + b, 0);

        if (!siloNo || totalWeight <= 0 || (link && link.length > 0)) continue;

        fillerDemands.push({
            recordId: record.id,
            siloNo,
            totalWeight,
            batchNum,
            cycle
        });
    }

    // 🧮 FIFO sort
    fillerDemands.sort((a, b) => {
        if (a.batchNum !== b.batchNum) return a.batchNum - b.batchNum;
        return a.cycle - b.cycle;
    });

    // 🗺️ Build silo map
    const siloMap = {};
    for (const silo of siloRecords.records) {
        const siloNo = getMatchValue(silo.getCellValue("Silo No"))?.toLowerCase();
        if (!siloNo) continue;
        if (!siloMap[siloNo]) siloMap[siloNo] = [];
        silo._remainingWeight = silo.getCellValue("Remaining Weight") || 0;
        siloMap[siloNo].push(silo);
    }

    for (const key in siloMap) {
        siloMap[key].sort((a, b) => (a.getCellValue("SILO Increment") || 0) - (b.getCellValue("SILO Increment") || 0));
    }

    // 🛠 Update collectors
    const siloUpdateMap = new Map();
    const mixerUpdateMap = new Map();

    for (const demand of fillerDemands) {
        const silos = siloMap[demand.siloNo.toLowerCase()];
        if (!silos || silos.length === 0) continue;

        let weightLeft = demand.totalWeight;
        const used = [];
        
        // Temporary storage for updates to ensure we only commit if demand is fully met
        const tempSiloUpdates = new Map();
        let tempWeightLeft = weightLeft;

        for (const silo of silos) {
            if (tempWeightLeft <= 0) break;
            const avail = silo._remainingWeight;
            if (avail <= 0) continue;

            const take = Math.min(avail, tempWeightLeft);
            const newRemaining = avail - take;
            
            tempWeightLeft -= take;
            used.push({ id: silo.id });

            // Stage the update locally
            tempSiloUpdates.set(silo.id, {
                id: silo.id,
                fields: { "Remaining Weight": newRemaining },
                _newVal: newRemaining // Store value to update memory object later
            });
        }

        // Only commit updates if the FULL weight requirement was met
        if (tempWeightLeft === 0 && used.length > 0) {
            mixerUpdateMap.set(demand.recordId, {
                id: demand.recordId,
                fields: { "Filler Silo ID": [...used] }
            });
            
            // Commit staged silo updates to the main update map
            for (const [siloId, updateData] of tempSiloUpdates) {
                siloUpdateMap.set(siloId, { 
                    id: updateData.id, 
                    fields: updateData.fields 
                });
                
                // Update the in-memory object so next iterations know this weight is gone
                const siloObj = silos.find(s => s.id === siloId);
                if (siloObj) {
                    siloObj._remainingWeight = updateData._newVal;
                }
            }
        }
    }

    // ✅ Apply silo weight updates
    const siloUpdates = Array.from(siloUpdateMap.values());
    while (siloUpdates.length > 0) {
        await siloTable.updateRecordsAsync(siloUpdates.slice(0, 50));
        siloUpdates.splice(0, 50);
    }

    // ✅ Apply mixer link updates
    const mixerUpdates = Array.from(mixerUpdateMap.values());
    while (mixerUpdates.length > 0) {
        await mixerTable.updateRecordsAsync(mixerUpdates.slice(0, 50));
        mixerUpdates.splice(0, 50);
    }

    console.log("✅ Filler Silo demand fulfilled using FIFO logic with batching.");
}

await allocateFillerSilo();
