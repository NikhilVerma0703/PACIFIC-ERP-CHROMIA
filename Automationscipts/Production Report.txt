// === UTILITIES ===
function resolveName(val) {
    if (!val) return "";
    if (Array.isArray(val)) {
        if (val.length === 0) return "";
        if (val[0] && typeof val[0] === 'object' && val[0].name) return val[0].name;
        return String(val[0]);
    }
    if (typeof val === 'object') return val.name || "";
    return String(val);
}

function normalizeString(s) {
    if (!s) return "";
    let str = s.toString().toLowerCase().trim();
    if (str.startsWith('c') && /^\d/.test(str.substring(1))) str = str.substring(1);
    return str.normalize('NFKD')
              .replace(/[\u0300-\u036f]/g, '')
              .replace(/[^a-z0-9\s]/g, '')
              .replace(/\s+/g, ' ')
              .trim();
}

function similarityScore(aRaw, bRaw) {
    const a = normalizeString(aRaw);
    const b = normalizeString(bRaw);
    if (!a && !b) return 0;
    if (a === b) return 1;
    if (a && b && (a.includes(b) || b.includes(a))) return 0.95;
    return 0;
}

function findBestPressMatch(pressList, mixerBatchName) {
    let bestIdx = -1;
    let bestScore = 0;
    let normMixer = normalizeString(mixerBatchName);

    for (let i = 0; i < pressList.length; i++) {
        const score = similarityScore(pressList[i].normalized, normMixer);
        if (score > bestScore) {
            bestScore = score;
            bestIdx = i;
        }
    }
    return bestScore >= 0.60 ? bestIdx : -1;
}
