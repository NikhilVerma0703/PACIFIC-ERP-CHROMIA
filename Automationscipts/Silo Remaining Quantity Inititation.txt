// Airtable-compatible wait (no setTimeout)
async function wait(ms) {
    const start = Date.now();
    while (Date.now() - start < ms) {
        await Promise.resolve();
    }
}

// Get input config
let { recordId } = input.config();

// Wait 15 seconds
await wait(10000);

// Access the SILO table and record
let siloTable = base.getTable("SILO");
let siloRecord = await siloTable.selectRecordAsync(recordId);

// Get linked record from 'Used Bag' field (expecting one)
let usedBagLinks = siloRecord.getCellValue("Used Bag");

if (usedBagLinks && usedBagLinks.length === 1) {
    let usedBagId = usedBagLinks[0].id;

    // Get the 'Used Bags' table
    let usedBagsTable = base.getTable("Used Bags");
    let usedBagRecord = await usedBagsTable.selectRecordAsync(usedBagId);

    // Extract Bag Weight
    let bagWeight = usedBagRecord.getCellValue("Bag Weight");

    if (typeof bagWeight === "number") {
        // Update 'Remaining Weight' and 'Weight' in the SILO record
        await siloTable.updateRecordAsync(recordId, {
            "Remaining Weight": bagWeight,
            "Weight": bagWeight
        });
    } else {
        console.error("Bag Weight is not a number.");
    }
} else {
    console.warn("No (or multiple) Used Bag record(s) linked.");
}
