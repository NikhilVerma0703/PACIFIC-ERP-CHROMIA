// Tables
let robotTable = base.getTable("Robot");
let mixerCycleTable = base.getTable("Mixer Cycle");

// Input
let inputConfig = input.config();
let createdRecordId = inputConfig.recordId;

// Fetch created Robot record
let robotRecord = await robotTable.selectRecordAsync(createdRecordId);
if (!robotRecord) {
    console.log("Record not found.");
    return;
}

// Check if it's the Last Slab
if (!robotRecord.getCellValue("Last Slab?")) {
    console.log("Not the last slab. Exiting.");
    return;
}

let batchName = robotRecord.getCellValue("Batch");
if (!batchName) {
    console.log("Batch missing. Exiting.");
    return;
}

// Fetch all slabs for this batch
let robotQuery = await robotTable.selectRecordsAsync({
    fields: ["Batch", "R2 Body Weight", "Slab Number", "Mixer Cycle"]
});

let slabs = robotQuery.records
    .filter(rec => rec.getCellValue("Batch") === batchName)
    .map(rec => ({
        id: rec.id,
        weight: rec.getCellValue("R2 Body Weight") || 0,
        slabNumber: rec.getCellValue("Slab Number") || 0
    }))
    .sort((a, b) => a.slabNumber - b.slabNumber);

// Fetch all mixer cycles for this batch
let mixerCycleQuery = await mixerCycleTable.selectRecordsAsync({
    fields: ["Batch", "Total Cycle Weight", "Cycle"]
});

let cycles = mixerCycleQuery.records
    .filter(rec => rec.getCellValue("Batch") === batchName)
    .map(rec => ({
        id: rec.id,
        cycleNumber: rec.getCellValue("Cycle") || 0,
        totalWeight: rec.getCellValue("Total Cycle Weight") || 0
    }))
    .sort((a, b) => a.cycleNumber - b.cycleNumber);

// Distribute slabs across mixer cycles
let updates = [];
let cycleIndex = 0;
let remainder = cycles[cycleIndex].totalWeight;

for (let i = 0; i < slabs.length; i++) {
    let slab = slabs[i];

    // If remainder < half the slab weight, move to next cycle
    if (remainder < (slab.weight / 2) && cycleIndex + 1 < cycles.length) {
        cycleIndex++;
        remainder = cycles[cycleIndex].totalWeight;
    }

    // Assign Mixer Cycle
    updates.push({
        id: slab.id,
        fields: {
            "Mixer Cycle": [{ id: cycles[cycleIndex].id }]
        }
    });

    // Subtract slab weight from remainder
    remainder -= slab.weight;
}

// Batch update slabs
while (updates.length > 0) {
    await robotTable.updateRecordsAsync(updates.slice(0, 50));
    updates = updates.slice(50);
}

console.log(`✅ Completed mixer cycle assignment for Batch: ${batchName}`);
