let inputConfig = input.config();
let recordId = inputConfig.recordId;

let table1 = base.getTable("Change Parameters Roy Mixer Cycle");
let rmTable = base.getTable("RM");

let rmLinkFields = ["F_RM", "G1_RM", "G2_RM", "G3_RM", "G4_RM", "G5_RM"];

// Fetch the new record
let newRecord = await table1.selectRecordAsync(recordId);
let linkedRmIds = new Set();

// Collect all linked RM record IDs
for (let field of rmLinkFields) {
    let linked = newRecord.getCellValue(field);
    if (linked && Array.isArray(linked)) {
        for (let rec of linked) {
            linkedRmIds.add(rec.id);
        }
    }
}

// Update each RM record to set Availability = "No"
let rmRecords = await rmTable.selectRecordsAsync();
let updates = [];

for (let id of linkedRmIds) {
    let rmRecord = rmRecords.getRecord(id);
    if (!rmRecord) continue;

    let currentAvailability = rmRecord.getCellValue("Availability");
    if (!currentAvailability || currentAvailability.name !== "No") {
        updates.push({
            id: id,
            fields: {
                "Availability": { name: "No" }
            }
        });
    }
}

// Batch update in chunks of 50
while (updates.length > 0) {
    await rmTable.updateRecordsAsync(updates.slice(0, 50));
    updates = updates.slice(50);
}
