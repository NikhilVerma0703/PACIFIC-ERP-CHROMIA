// Tables
let kreosTable = base.getTable("Kreos");
let mixerCycleTable = base.getTable("Mixer Cycle");
let wastageTable = base.getTable("wastage");
let changeParamsTable = base.getTable("Change Parameters Kreos");

// Input
let inputConfig = input.config();
let createdRecordId = inputConfig.recordId;

// 🔧 Normalize helper
function normalize(value) {
    if (!value) return "";
    return String(value).toLowerCase().replace(/\s+/g, "");
}

// Get created record
let kreosRecord = await kreosTable.selectRecordAsync(createdRecordId);
if (!kreosRecord || !kreosRecord.getCellValue("Last Slab?")) {
    console.log("⛔ Not the last slab. Exiting script.");
    return;
}

let currentSlabNumber = kreosRecord.getCellValue("Slab Number");
if (currentSlabNumber == null) {
    console.log("❌ Missing Slab Number. Exiting.");
    return;
}

// 🔍 Get Batch from "Change Parameters Kreos"
let changeRecords = await changeParamsTable.selectRecordsAsync({
    fields: ["Effective from slab number", "Batch"]
});

let matchingRecord = changeRecords.records
    .filter(r => {
        let eff = r.getCellValue("Effective from slab number");
        return eff != null && eff <= currentSlabNumber;
    })
    .sort((a, b) =>
        b.getCellValue("Effective from slab number") - a.getCellValue("Effective from slab number")
    )[0];

if (!matchingRecord) {
    console.log("❌ No matching batch found in 'Change Parameters Kreos'. Exiting.");
    return;
}

let rawBatch = matchingRecord.getCellValue("Batch");
if (!rawBatch) {
    console.log("❌ Missing Batch in selected Change Parameters record. Exiting.");
    return;
}
let batchName = normalize(rawBatch);
console.log(`🎯 Using normalized batch "${batchName}" from Change Parameters Kreos`);

// Step 2: Get All Slabs for the Batch
let kreosQuery = await kreosTable.selectRecordsAsync({
    fields: ["Batch", "Slab Number", "Slab Weight", "Chessboard Body Percentage"]
});
let slabs = kreosQuery.records
    .filter(rec => normalize(rec.getCellValue("Batch")) === batchName)
    .sort((a, b) => (a.getCellValue("Slab Number") || 0) - (b.getCellValue("Slab Number") || 0));

// Step 3: Fetch Mixer Cycles for the Batch
let mixerQuery = await mixerCycleTable.selectRecordsAsync({
    fields: ["Batch", "Cycle", "Total Cycle Weight", "Total Mixer 1 Weight"]
});
let mixerCycles = mixerQuery.records
    .filter(rec => normalize(rec.getCellValue("Batch")) === batchName);

// Step 4: Build slabWeightsMap using per‑slab Chessboard %
let slabWeightsMap = new Map();
for (let slab of slabs) {
    let slabWeight = slab.getCellValue("Slab Weight") || 0;
    let pct = slab.getCellValue("Chessboard Body Percentage") || 0;

    let chessboardWeight = (pct / 100) * slabWeight;
    let normalWeight = slabWeight - chessboardWeight;

    slabWeightsMap.set(slab.id, { chessboardWeight, normalWeight });
}

// Step 5: Calculate overall Wastage %
let totalCycleWeight = 0;
let totalMixer1Weight = 0;
for (let c of mixerCycles) {
    totalCycleWeight += c.getCellValue("Total Cycle Weight") || 0;
    totalMixer1Weight += c.getCellValue("Total Mixer 1 Weight") || 0;
}
let totalOtherWeight = totalCycleWeight - totalMixer1Weight;

let sumChessWeight = 0, sumNormalWeight = 0;
for (let slab of slabs) {
    let { chessboardWeight, normalWeight } = slabWeightsMap.get(slab.id);
    sumChessWeight += chessboardWeight;
    sumNormalWeight += normalWeight;
}

let mixer1WastagePercent = totalMixer1Weight === 0 
    ? 0 
    : (totalMixer1Weight - sumChessWeight) / totalMixer1Weight;

let normalWastagePercent = totalOtherWeight === 0 
    ? 0 
    : (totalOtherWeight - sumNormalWeight) / totalOtherWeight;

// Step 6: Apply %‑wastage to cycle weights
let m1Cycles = [], normalCycles = [];
for (let c of mixerCycles) {
    let id = c.id;
    let total = c.getCellValue("Total Cycle Weight") || 0;
    let m1 = c.getCellValue("Total Mixer 1 Weight") || 0;
    let oth = total - m1;

    let adjM1 = m1 * (1 - mixer1WastagePercent);
    let adjOth = oth * (1 - normalWastagePercent);

    if (adjM1 > 0) m1Cycles.push({ id, adjusted: adjM1 });
    if (adjOth > 0) normalCycles.push({ id, adjusted: adjOth });
}

// Sort FIFO by Cycle number
let cycleMap = new Map(mixerCycles.map(c => [c.id, c]));
let getCycleIndex = id => cycleMap.get(id).getCellValue("Cycle") || 0;

m1Cycles.sort((a, b) => getCycleIndex(a.id) - getCycleIndex(b.id));
normalCycles.sort((a, b) => getCycleIndex(a.id) - getCycleIndex(b.id));

// Step 7: Apply explicit wastage records
let wastQ = await wastageTable.selectRecordsAsync({
    fields: ["Batch", "Slab number after which material dumped", "Amount wasted"]
});
let wastRecs = wastQ.records.filter(r => normalize(r.getCellValue("Batch")) === batchName);

for (let w of wastRecs) {
    let afterNum = w.getCellValue("Slab number after which material dumped");
    let amt = w.getCellValue("Amount wasted") || 0;

    let idx = slabs.findIndex(s => (s.getCellValue("Slab Number") || 0) === afterNum);
    if (idx >= 0 && idx + 1 < slabs.length) {
        let next = slabs[idx + 1];
        let { chessboardWeight, normalWeight } = slabWeightsMap.get(next.id);
        let total = chessboardWeight + normalWeight;
        if (total > 0) {
            let wM1 = (chessboardWeight / total) * amt;
            let wNor = (normalWeight / total) * amt;
            if (m1Cycles[0])     m1Cycles[0].adjusted = Math.max(0, m1Cycles[0].adjusted - wM1);
            if (normalCycles[0]) normalCycles[0].adjusted = Math.max(0, normalCycles[0].adjusted - wNor);
        }
    }
}

// Step 8: Assign cycles (FIFO) to each slab
let updates = [];
let m1Idx = 0, remM1 = m1Cycles[0]?.adjusted || 0;
let normIdx = 0, remNor = normalCycles[0]?.adjusted || 0;

for (let slab of slabs) {
    let { chessboardWeight, normalWeight } = slabWeightsMap.get(slab.id);

    while (remNor < normalWeight && normIdx < normalCycles.length - 1) {
        normIdx++; remNor += normalCycles[normIdx].adjusted;
    }
    let normId = normalCycles[normIdx]?.id || null;
    remNor -= normalWeight;

    while (remM1 < chessboardWeight && m1Idx < m1Cycles.length - 1) {
        m1Idx++; remM1 += m1Cycles[m1Idx].adjusted;
    }
    let m1Id = m1Cycles[m1Idx]?.id || null;
    remM1 -= chessboardWeight;

    updates.push({
        id: slab.id,
        fields: {
            "Mixer Cycle": normId ? [{ id: normId }] : [],
            "Chessboard Mixer Cycle": m1Id ? [{ id: m1Id }] : []
        }
    });
}

// Step 9: Push updates in batches
while (updates.length) {
    await kreosTable.updateRecordsAsync(updates.slice(0, 50));
    updates = updates.slice(50);
}

console.log("✅ Finished assigning both Mixer Cycle and Chessboard Mixer Cycle per-slab.");
