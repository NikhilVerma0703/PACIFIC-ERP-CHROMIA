// Tables
let jotTable = base.getTable("Jot");

// --- CONFIGURATION: FIELD NAMES (UPDATED) ---
// Both tables use lowercase 'n' for "Design name"
const PRESS_FIELD_BATCH = "Batch";
const PRESS_FIELD_DESIGN = "Design name"; // Corrected based on error log
const JOT_FIELD_BATCH = "Batch";
const JOT_FIELD_DESIGN = "Design name";   
// --------------------------------------------

// Safety Check: Slab Summary
let slabSummaryTable = null;
try {
    slabSummaryTable = base.getTable("Slab Summary");
} catch (e) {
    console.warn("Table 'Slab Summary' not found. Ending script silently.");
    return;
}

// Get input record ID
let inputConfig = input.config();
let jotRecordId = inputConfig.recordId;

// Load the Jot record (Current Trigger)
let jotRecord = await jotTable.selectRecordAsync(jotRecordId);
let slabNumberRaw = jotRecord.getCellValue("Slab Number");

if (!slabNumberRaw) throw new Error("Slab Number is missing in the Jot record.");
let slabNumber = parseFloat(slabNumberRaw);
if (isNaN(slabNumber)) throw new Error(`Slab Number "${slabNumberRaw}" is not a valid number.`);

// --- HELPER: Extract Simple Text ---
function extractValue(record, fieldName) {
    let val = null;
    try { val = record.getCellValue(fieldName); } catch (e) { return null; }
    if (!val) return null;
    if (Array.isArray(val)) return val.map(item => item.name || item).join(", ");
    if (typeof val === 'object' && val.name) return val.name;
    return val;
}

// =================================================================================
// PART 1: ORIGINAL SUMMARY CREATION LOGIC (Preserved)
// =================================================================================

let directMatchTables = [
    { name: "Kreos", linkField: "Kreos", hasMixer: true },
    { name: "Distributor", linkField: "Distributor", hasMixer: true },
    { name: "Press", linkField: "Press", hasMixer: false },
    { name: "Oven", linkField: "Oven", hasMixer: false },
    { name: "Jot", linkField: "Jot", hasMixer: false },
    { name: "Robot", linkField: "Robot", hasMixer: true }
];

let royMixerLinkField = "Roy Mixer";
let updates = { "Slab Number": slabNumber };
let mixerCycleId = null;
let mixerCycleRecord = null;

// Loop through tables
for (let tableInfo of directMatchTables) {
    let table = null;
    try { table = base.getTable(tableInfo.name); } catch (e) { continue; }

    let fields = ["Slab Number"];
    if (tableInfo.hasMixer) fields.push("Mixer Cycle");

    let records = await table.selectRecordsAsync({ fields });
    let match = records.records.find(rec => {
        let val = rec.getCellValue("Slab Number");
        return parseFloat(val) === slabNumber;
    });

    if (match) {
        updates[tableInfo.linkField] = [{ id: match.id }];

        if (tableInfo.name === "Robot") {
            let royMixerLink = match.getCellValue("Mixer Cycle");
            if (royMixerLink && royMixerLink.length > 0) {
                updates[royMixerLinkField] = [{ id: royMixerLink[0].id }];
            }
        }

        if (tableInfo.hasMixer && !mixerCycleId) {
            let mixerLink = match.getCellValue("Mixer Cycle");
            if (mixerLink && mixerLink.length > 0) {
                mixerCycleId = mixerLink[0].id;
            }
        }
    }
}

// Add Mixer Cycle link if found
if (mixerCycleId) updates["Mixer Cycle"] = [{ id: mixerCycleId }];

// Deep dive into Mixer Cycle
if (mixerCycleId) {
    let mixerCycleTable = null;
    try { mixerCycleTable = base.getTable("Mixer Cycle"); } catch (e) {}

    if (mixerCycleTable) {
        mixerCycleRecord = await mixerCycleTable.selectRecordAsync(mixerCycleId);

        function collectUniqueLinks(fieldNames) {
            let linkMap = new Map();
            for (let field of fieldNames) {
                let cell = mixerCycleRecord.getCellValue(field);
                if (cell && Array.isArray(cell)) {
                    for (let rec of cell) linkMap.set(rec.id, rec);
                }
            }
            return Array.from(linkMap.values());
        }

        let gritFields = [];
        for (let i = 1; i <= 4; i++) {
            for (let j = 1; j <= 5; j++) gritFields.push(`M${i}_G${j}`);
        }
        let gritLinks = collectUniqueLinks(gritFields);
        if (gritLinks.length > 0) updates["Grit Links"] = gritLinks;

        let fillerLookup = mixerCycleRecord.getCellValue("Filler Silo ID");
        if (fillerLookup && Array.isArray(fillerLookup)) updates["Filler Links"] = [...fillerLookup];

        let resinFields = ["M1_R_ID", "M2_R_ID", "M3_R_ID", "M4_R_ID"];
        let resinLinks = collectUniqueLinks(resinFields);
        if (resinLinks.length > 0) updates["Resin Links"] = resinLinks;
    }
}

// Create Slab Summary record
await slabSummaryTable.createRecordAsync(updates);
console.log(`✅ Slab Summary created for Slab #${slabNumber}`);


// =================================================================================
// PART 2: GLOBAL BACKFILL LOGIC (Corrected Field Names)
// =================================================================================

console.log("🔄 Starting Global Backfill for missing Batch/Design data...");

let pressTable = null;
try { pressTable = base.getTable("Press"); } catch(e) {}

if (pressTable) {
    // 1. Create a Reference Map of ALL Press Data
    // We fetch all records to ensure we can match any slab number
    let pressQuery = await pressTable.selectRecordsAsync({
        fields: ["Slab Number", PRESS_FIELD_BATCH, PRESS_FIELD_DESIGN]
    });

    let pressMap = new Map();
    
    for (let rec of pressQuery.records) {
        let sNum = parseFloat(rec.getCellValue("Slab Number"));
        if (!isNaN(sNum)) {
            let pBatch = extractValue(rec, PRESS_FIELD_BATCH);
            let pDesign = extractValue(rec, PRESS_FIELD_DESIGN);
            
            // Only store if there is useful data
            if (pBatch || pDesign) {
                pressMap.set(sNum, { batch: pBatch, design: pDesign });
            }
        }
    }
    console.log(`📊 Indexed ${pressMap.size} Press records with valid data.`);

    // 2. Scan ALL Jot Records for missing fields
    let jotQuery = await jotTable.selectRecordsAsync({
        fields: ["Slab Number", JOT_FIELD_BATCH, JOT_FIELD_DESIGN]
    });

    let updatesToProcess = [];

    for (let rec of jotQuery.records) {
        let currentBatch = extractValue(rec, JOT_FIELD_BATCH);
        let currentDesign = extractValue(rec, JOT_FIELD_DESIGN);

        // Check if this record is "incomplete"
        if (!currentBatch || !currentDesign) {
            let sNum = parseFloat(rec.getCellValue("Slab Number"));
            
            if (!isNaN(sNum) && pressMap.has(sNum)) {
                let matchData = pressMap.get(sNum);
                let recordUpdates = {};

                // Only update if Jot is empty AND Press has data
                if (!currentBatch && matchData.batch) {
                    recordUpdates[JOT_FIELD_BATCH] = matchData.batch;
                }
                if (!currentDesign && matchData.design) {
                    recordUpdates[JOT_FIELD_DESIGN] = matchData.design;
                }

                // If we found something to update, add to queue
                if (Object.keys(recordUpdates).length > 0) {
                    updatesToProcess.push({
                        id: rec.id,
                        fields: recordUpdates
                    });
                }
            }
        }
    }

    // 3. Execute Batch Updates (Max 50 at a time)
    if (updatesToProcess.length > 0) {
        console.log(`⚡ Found ${updatesToProcess.length} Jot records to backfill.`);
        
        while (updatesToProcess.length > 0) {
            let batch = updatesToProcess.splice(0, 50);
            await jotTable.updateRecordsAsync(batch);
        }
        console.log("✅ All backfill updates completed.");
    } else {
        console.log("ℹ️ No matched backfill data found (either Jot is full or Press is empty).");
    }

} else {
    console.warn("⚠️ Press table not found, skipping backfill.");
}
